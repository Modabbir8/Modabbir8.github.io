placement_rank.sqlAthena / Trino
-- Rank each customer's placements and flag orders after the 5th
WITH ranked AS (
  SELECT customer_id, order_id, order_created_at,
         ROW_NUMBER() OVER (PARTITION BY customer_id
                            ORDER BY order_created_at) AS placement_rank
  FROM placements
  WHERE is_genuine_placement
)
SELECT customer_id,
       COUNT_IF(placement_rank > 5) AS repeat_placements,
       DATE_TRUNC('month', MIN(order_created_at)) AS cohort_month
FROM ranked
GROUP BY customer_id;
